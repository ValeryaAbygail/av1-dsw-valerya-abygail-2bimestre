                        🔄 FLUXO DE DADOS DA APLICAÇÃO

    Cliente HTTP                      Express API                    Banco
    ─────────────                     ────────────                   ─────

    GET /tarefas                 
       ├──────────────────────────→ Router
       │                                 └──→ Controller.listar()
       │                                      └──→ Model.listar()
       │                                           └──→ Prisma.findMany()
       │                                                └──→ SELECT * FROM Task
       │                                ←──────────── Retorna tasks com categories
       └─ JSON {sucesso, dados}


    POST /tarefas {title, desc}
       ├──────────────────────────→ Router
       │                                 └──→ Controller.criar()
       │                                      ├─ Valida title
       │                                      ├─ Valida description
       │                                      └─→ Model.criar()
       │                                           └──→ Prisma.create()
       │                                                └──→ INSERT INTO Task
       │                                ←──────────── 201 + novo registro
       └─ JSON {sucesso, mensagem, dados}


    PUT /tarefas/1 {completed}
       ├──────────────────────────→ Router
       │                                 └──→ Controller.atualizar()
       │                                      ├─ Valida ID
       │                                      ├─ Valida completed
       │                                      └─→ Model.atualizar()
       │                                           └──→ Prisma.update()
       │                                                └──→ UPDATE Task SET...
       │                                ←──────────── Retorna updated task
       └─ JSON {sucesso, mensagem, dados}


    DELETE /tarefas/1
       ├──────────────────────────→ Router
       │                                 └──→ Controller.excluir()
       │                                      ├─ Valida ID
       │                                      └─→ Model.excluir()
       │                                           └──→ Prisma.delete()
       │                                                └──→ DELETE FROM Task
       │                                ←──────────── Retorna deleted task
       └─ JSON {sucesso, mensagem, dados}
